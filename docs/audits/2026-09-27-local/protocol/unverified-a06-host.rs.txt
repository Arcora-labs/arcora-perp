//! Execute frozen deterministic parity vectors in the real SP1 RISC-V executor.
//!
//! Native evaluation and guest input use the same postcard bytes. This executable
//! checks full public commitments (seven roots, plus the A06 phase word), actual
//! guest failures, and external expected-commitment mismatches. It does not prove
//! or submit anything, and makes no network-prover request.

use k256::ecdsa::SigningKey;
use perp_core::commitment::{derive_roots, DerivedRoots};
use perp_core::engine::BatchOp;
use perp_core::fixed::{PRICE_SCALE, QUOTE_SCALE, SIZE_SCALE};
use perp_core::hash::{word_u64, Domain, Hasher, Keccak256};
use perp_core::market::Market;
use perp_core::note::{owner_from_spend_key, Note};
use perp_core::oracle::{oracle_digest, OracleSig, OracleTranscript};
use perp_core::order::{BatchManifest, Side};
use perp_core::{DefaultState, EngineError};
use sp1_sdk::{include_elf, CpuProver, Elf, Prover, ProverClient, SP1Stdin};

const ELF: Elf = include_elf!("perp-core-guest");
type Witness = (DefaultState, Vec<BatchOp>, BatchManifest);

fn manifest(state: &DefaultState) -> BatchManifest {
    BatchManifest {
        previous_state_root: state.state_root(),
        batch_id: state.next_batch_id,
        ordered: vec![],
        rejected: vec![],
        oracle_updates: vec![],
        matching_rule_version: 0,
        enclave_measurement: [0; 32],
        sequencer_pubkey_epoch: 0,
    }
}

fn witness(state: DefaultState, ops: Vec<BatchOp>) -> Witness {
    let manifest = manifest(&state);
    (state, ops, manifest)
}

fn deposit_withdraw() -> Witness {
    let spend = [3; 32];
    let owner = owner_from_spend_key::<Keccak256>(&spend);
    let blind = [9; 32];
    let amount = QUOTE_SCALE;
    let mut state = DefaultState::new(16);
    state.add_market(Market::conservative(0));
    witness(
        state,
        vec![
            BatchOp::Deposit {
                owner,
                asset_id: 0,
                amount,
                blinding: blind,
                from: [0; 20],
                deposit_id: 0,
                deposit_blind: [0; 32],
            },
            BatchOp::Withdraw {
                note_commitment: Note::new(owner, 0, amount, blind).commitment::<Keccak256>(),
                spend_key: spend,
                to: Some([0xAB; 20]),
                nonce: 1,
            },
        ],
    )
}

fn oracle(now: u64) -> OracleTranscript {
    let price = 100_000 * PRICE_SCALE;
    let digest = oracle_digest(0, price, now, PRICE_SCALE, price);
    OracleTranscript {
        price,
        publish_time_ms: now,
        confidence: PRICE_SCALE,
        backup_twap: price,
        signature: OracleSig::sign(&SigningKey::from_slice(&[7; 32]).unwrap(), &digest),
    }
}

/// Adapted from perp-core/tests/lifecycle.rs; real publisher signatures, two
/// counterparties, funded margins, a nonzero funding mark spread and withdrawal.
fn trade_funding_withdraw() -> Witness {
    let mut state = DefaultState::new(16);
    let mut market = Market::conservative(0);
    let signed = oracle(1_000);
    market.oracle_pubkey = signed
        .signature
        .recover(&oracle_digest(
            0,
            signed.price,
            signed.publish_time_ms,
            signed.confidence,
            signed.backup_twap,
        ))
        .unwrap();
    state.add_market(market);
    let mut ops = vec![];
    let owners = [
        owner_from_spend_key::<Keccak256>(&[1; 32]),
        owner_from_spend_key::<Keccak256>(&[2; 32]),
    ];
    for (index, owner) in owners.iter().enumerate() {
        let blind = [0x11 + index as u8; 32];
        ops.push(BatchOp::Deposit {
            owner: *owner,
            asset_id: 0,
            amount: 20_000 * QUOTE_SCALE,
            blinding: blind,
            from: [0xA1 + index as u8; 20],
            deposit_id: index as u64,
            deposit_blind: [0xDB; 32],
        });
        ops.push(BatchOp::FundPosition {
            owner: *owner,
            market_id: 0,
            note_commitment: Note::new(*owner, 0, 20_000 * QUOTE_SCALE, blind)
                .commitment::<Keccak256>(),
            spend_key: [index as u8 + 1; 32],
        });
    }
    ops.push(BatchOp::Fill {
        taker: owners[0],
        maker: owners[1],
        market_id: 0,
        taker_side: Side::Buy,
        size: SIZE_SCALE,
        price: 100_000 * PRICE_SCALE,
        oracle: oracle(1_000),
        now_ms: 1_000,
    });
    ops.push(BatchOp::AccrueFunding {
        market_id: 0,
        mark: 100_001 * PRICE_SCALE,
        oracle: oracle(2_000),
        now_ms: 2_000,
    });
    let blind = [0x33; 32];
    ops.push(BatchOp::Unbind {
        owner: owners[0],
        market_id: 0,
        amount: 1_000 * QUOTE_SCALE,
        blinding: blind,
        oracle: oracle(3_000),
        now_ms: 3_000,
    });
    ops.push(BatchOp::Withdraw {
        note_commitment: Note::new(owners[0], 0, 1_000 * QUOTE_SCALE, blind)
            .commitment::<Keccak256>(),
        spend_key: [1; 32],
        to: Some([0xAA; 20]),
        nonce: 7,
    });
    witness(state, ops)
}

fn serialized(name: &str, input: &Witness) -> (Witness, SP1Stdin) {
    let bytes = postcard::to_allocvec(input).expect("witness encode");
    let decoded: Witness = postcard::from_bytes(&bytes).expect("same witness decode");
    assert_eq!(
        postcard::to_allocvec(&decoded).unwrap(),
        bytes,
        "{name}: byte-exact roundtrip"
    );
    save(name, "witness.bin", &bytes);
    println!(
        "PARITY_INPUT name={name} bytes={} roundtrip=true",
        bytes.len()
    );
    let mut stdin = SP1Stdin::new();
    stdin.write_vec(bytes);
    (decoded, stdin)
}

fn save(name: &str, suffix: &str, bytes: &[u8]) {
    if let Some(path) = std::env::var_os("ARCORA_PARITY_OUTPUT") {
        let path = std::path::PathBuf::from(path);
        std::fs::create_dir_all(&path).expect("fixture output directory");
        std::fs::write(path.join(format!("{name}.{suffix}")), bytes).expect("fixture output write");
    }
}

fn consumer_preimage(roots: &DerivedRoots) -> Vec<u8> {
    // DarkPerpSettlement.publicCommitment / windDownCommitment: domain byte,
    // seven roots in this order, and a little-endian 32-byte nonzero phase word.
    let mut words = vec![
        roots.prev_state_root,
        roots.manifest_hash,
        roots.new_state_root,
        roots.ordered_root,
        roots.withdrawals_root,
        roots.rejected_root,
        roots.deposits_root,
    ];
    if roots.wind_down_phase != 0 {
        words.push(word_u64(roots.wind_down_phase as u64));
    }
    assert_eq!(
        Keccak256::hash_words(Domain::StateRoot, &words),
        roots.commitment::<Keccak256>()
    );
    let mut bytes = vec![Domain::StateRoot as u8];
    for word in words {
        bytes.extend_from_slice(&word);
    }
    bytes
}

async fn valid(
    client: &CpuProver,
    name: &str,
    input: &Witness,
    phase: u8,
) -> (DefaultState, DerivedRoots) {
    let ((mut state, ops, manifest), stdin) = serialized(name, input);
    let roots = derive_roots(&mut state, &ops, &manifest).expect("valid native transition");
    assert_eq!(roots.wind_down_phase, phase);
    assert!(state.conservation_holds());
    let native = roots.commitment::<Keccak256>();
    let (public, report) = client
        .execute(ELF, stdin)
        .await
        .expect("real guest executes");
    let guest: [u8; 32] = public
        .as_slice()
        .try_into()
        .expect("exactly 32 public bytes");
    assert_eq!(
        native, guest,
        "{name}: full native/guest commitment equality"
    );
    save(name, "public.bin", &guest);
    save(name, "consumer-preimage.bin", &consumer_preimage(&roots));
    println!(
        "PARITY_VALID name={name} phase={phase} cycles={} native={} guest={}",
        report.total_instruction_count(),
        hex(&native),
        hex(&guest)
    );
    println!("PARITY_ROOTS name={name} prev={} manifest={} post={} ordered={} withdrawals={} rejected={} deposits={}",
        hex(&roots.prev_state_root), hex(&roots.manifest_hash), hex(&roots.new_state_root), hex(&roots.ordered_root),
        hex(&roots.withdrawals_root), hex(&roots.rejected_root), hex(&roots.deposits_root));
    (state, roots)
}

async fn rejected(client: &CpuProver, name: &str, input: &Witness, expected: EngineError) {
    let ((mut state, ops, manifest), stdin) = serialized(name, input);
    assert_eq!(
        derive_roots(&mut state, &ops, &manifest).unwrap_err(),
        expected,
        "{name}: native rejects for the intended reason"
    );
    // The native rejection does NOT short-circuit guest execution: this exact
    // serialized invalid witness is deliberately sent into the real executor.
    let error = match client.execute(ELF, stdin).await {
        Ok(_) => panic!("{name}: invalid witness unexpectedly executed in the guest"),
        Err(error) => error,
    };
    let message = error.to_string();
    assert!(
        message.contains("exit code") || message.contains("Exited") || message.contains("panic"),
        "{name}: an infrastructure error is not guest rejection: {message}"
    );
    println!(
        "PARITY_REJECT name={name} native={expected:?} guest_reached=true sdk_error={message}"
    );
}

#[tokio::main]
async fn main() {
    let client = ProverClient::builder().cpu().build().await;
    let ordinary = deposit_withdraw();
    let (_, ordinary_roots) = valid(&client, "normal-deposit-withdraw", &ordinary, 0).await;
    let trading = trade_funding_withdraw();
    let (traded, _) = valid(&client, "normal-trade-funding-withdraw", &trading, 0).await;
    assert!(traded.positions.values().any(|position| position.size != 0));
    let phase1 = witness(traded, vec![BatchOp::SettleAll]);
    let (closed, roots1) = valid(&client, "a06-phase1", &phase1, 1).await;
    assert!(closed.positions.values().all(|position| position.size == 0));
    let owner = owner_from_spend_key::<Keccak256>(&[1; 32]);
    let blind = [0xA6; 32];
    let amount = 100 * QUOTE_SCALE;
    let phase2 = witness(
        closed,
        vec![
            BatchOp::WindDownUnbind {
                owner,
                market_id: 0,
                amount,
                blinding: blind,
            },
            BatchOp::WindDownWithdraw {
                note_commitment: Note::new(owner, 0, amount, blind).commitment::<Keccak256>(),
                spend_key: [1; 32],
                to: Some([0x44; 20]),
                nonce: 9,
            },
        ],
    );
    assert_eq!(
        phase2.2.previous_state_root, roots1.new_state_root,
        "phase transition continuity"
    );
    let (_, roots2) = valid(&client, "a06-phase2", &phase2, 2).await;

    let mut bad = ordinary.clone();
    bad.2.previous_state_root[0] ^= 1;
    rejected(
        &client,
        "wrong-previous-root",
        &bad,
        EngineError::ManifestMismatch,
    )
    .await;
    let mut bad = ordinary.clone();
    bad.0.consumed_deposit_tip[0] ^= 1;
    rejected(
        &client,
        "tampered-deposit-prefix",
        &bad,
        EngineError::ManifestMismatch,
    )
    .await;
    let mut bad = ordinary.clone();
    if let BatchOp::Deposit { deposit_id, .. } = &mut bad.1[0] {
        *deposit_id = 1;
    }
    rejected(
        &client,
        "skipped-deposit-id",
        &bad,
        EngineError::DepositOutOfOrder,
    )
    .await;
    let mut bad = ordinary.clone();
    if let BatchOp::Withdraw { spend_key, .. } = &mut bad.1[1] {
        *spend_key = [4; 32];
    }
    rejected(
        &client,
        "wrong-spend-authority",
        &bad,
        EngineError::BadSpendKey,
    )
    .await;
    let mut bad = phase1.clone();
    bad.1.push(phase2.1[0].clone());
    rejected(
        &client,
        "mixed-wind-down-phase",
        &bad,
        EngineError::WindDownGrammar,
    )
    .await;
    let mut bad = ordinary.clone();
    bad.0.next_batch_id = u64::MAX;
    bad.2 = manifest(&bad.0);
    rejected(
        &client,
        "exhausted-batch-counter",
        &bad,
        EngineError::Overflow,
    )
    .await;

    // Deposit identity is bound into the public commitment; external L1 truth is
    // not a private guest input. Altering a well-formed event can execute, but it
    // MUST NOT satisfy the original expected public commitment.
    let mut changed = ordinary.clone();
    if let BatchOp::Deposit { deposit_blind, .. } = &mut changed.1[0] {
        deposit_blind[0] ^= 1;
    }
    let (_, changed_roots) = valid(&client, "changed-deposit-commitment", &changed, 0).await;
    assert_ne!(changed_roots.deposits_root, ordinary_roots.deposits_root);
    assert_ne!(
        changed_roots.commitment::<Keccak256>(),
        ordinary_roots.commitment::<Keccak256>()
    );
    println!(
        "PARITY_BINDING name=tampered-deposit-commitment expected_mismatch=true guest_error=false"
    );
    let mut wrong_phase = roots2;
    wrong_phase.wind_down_phase = 1;
    assert_ne!(
        wrong_phase.commitment::<Keccak256>(),
        roots2.commitment::<Keccak256>()
    );
    println!(
        "PARITY_BINDING name=tampered-phase-public-input expected_mismatch=true guest_error=false"
    );
    println!("PARITY_DONE valid_guest_executions=5 invalid_guest_rejections=6 binding_negatives=2 mocked=0");
}

fn hex(bytes: &[u8]) -> String {
    bytes.iter().map(|byte| format!("{byte:02x}")).collect()
}
